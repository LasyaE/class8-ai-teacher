# 2. Dataset Validation
input_dir = "data\\extracted\\raw"
output_dir = "data\\processed\\jsonl"
train_files = []
val_files = []
test_files = []

for subject in os.listdir(input_dir):
    subject_path = os.path.join(input_dir, subject)
    if not os.path.isdir(subject_path):
        continue
    files = os.listdir(subject_path)
    files.sort()
    train_split = files[:int(0.8 * len(files))]
    val_split = files[int(0.8 * len(files)):int(0.9 * len(files))]
    test_split = files[int(0.9 * len(files)):]

    for file in train_split:
        file_path = os.path.join(subject_path, file)
        with open(file_path, "r", encoding="utf-8") as f:
            content = f.read()
            lines = content.split("\n")
            metadata = re.match(r"Chapter \d+, Page \d+, Source: .+", lines[0])
            if not metadata:
                continue
            text = "\n".join(lines[1:])
            qa_pairs = re.findall(r"Question: (.+?)\nAnswer: (.+?)\n", text)
            for q, a in qa_pairs:
                train_files.append({"instruction": f"Answer based on {metadata.group(0).split("Source: ")[1]}", "input": q, "output": a, "subject": subject, "chapter": metadata.group(0).split("Chapter ")[1].split(",")[0], "page": metadata.group(0).split("Page ")[1].split(",")[0], "source": metadata.group(0).split("Source: ")[1]})

    for file in val_split:
        file_path = os.path.join(subject_path, file)
        with open(file_path, "r", encoding="utf-8") as f:
            content = f.read()
            lines = content.split("\n")
            metadata = re.match(r"Chapter \d+, Page \d+, Source: .+", lines[0])
            if not metadata:
                continue
            text = "\n".join(lines[1:])
            qa_pairs = re.findall(r"Question: (.+?)\nAnswer: (.+?)\n", text)
            for q, a in qa_pairs:
                val_files.append({"instruction": f"Answer based on {metadata.group(0).split("Source: ")[1]}", "input": q, "output": a, "subject": subject, "chapter": metadata.group(0).split("Chapter ")[1].split(",")[0], "page": metadata.group(0).split("Page ")[1].split(",")[0], "source": metadata.group(0).split("Source: ")[1]})

    for file in test_split:
        file_path = os.path.join(subject_path, file)
        with open(file_path, "r", encoding="utf-8") as f:
            content = f.read()
            lines = content.split("\n")
            metadata = re.match(r"Chapter \d+, Page \d+, Source: .+", lines[0])
            if not metadata:
                continue
            text = "\n".join(lines[1:])
            qa_pairs = re.findall(r"Question: (.+?)\nAnswer: (.+?)\n", text)
            for q, a in qa_pairs:
                test_files.append({"instruction": f"Answer based on {metadata.group(0).split("Source: ")[1]}", "input": q, "output": a, "subject": subject, "chapter": metadata.group(0).split("Chapter ")[1].split(",")[0], "page": metadata.group(0).split("Page ")[1].split(",")[0], "source": metadata.group(0).split("Source: ")[1]})
